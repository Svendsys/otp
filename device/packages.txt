# Apt packages the print unit needs, one per line: the single source for
# device/install.sh and the pi-gen stage. All are baked into the image, so
# first boot needs no network.

# Generation
python3-reportlab

# Front panel: 128x64 SSD1306 OLED over I2C, three buttons on GPIO
python3-luma.oled
python3-pil
python3-gpiozero
# From archive.raspberrypi.org, not Debian proper, so install.sh needs
# Raspberry Pi OS rather than plain Debian.
python3-lgpio
i2c-tools
# otpunit/diagnostics.py imports smbus2 directly for the status sheet's I2C
# scan, so it is listed rather than left to python3-luma.core's Depends.
python3-smbus2

# Printing. ipp-usb covers driverless IPP-over-USB, which is most printers
# made since roughly 2017.
cups
cups-client
cups-filters
ipp-usb
ghostscript

# Drivers, listed individually: printer-driver-all carries them only under
# Recommends, so with --no-install-recommends it installs nothing.
# HP LaserJet Pro M12a/M12w and the rest of the ZjStream family
printer-driver-foo2zjs
# Brother HL-2030/2140/L2300 and relatives
printer-driver-brlaser
# Broad coverage, including many PCL lasers
printer-driver-gutenprint
# HP's own raster driver
printer-driver-hpcups
# PostScript HP models
printer-driver-postscript-hp
# Samsung and Xerox SPL
printer-driver-splix
# Minolta PagePro
printer-driver-min12xxw

# The Foomatic PPD database, precompiled: host-based lasers need a PPD, not
# a driverless queue. Not foomatic-db, which Conflicts with this package.
foomatic-db-compressed-ppds
foomatic-db-engine


# Entropy. The Pi's TRNG driver (bcm2835-rng) is built into the kernel and
# seeds the CRNG before userspace exists; rngd only keeps feeding /dev/hwrng
# into the pool afterwards, so nothing in the boot path may depend on it.
rng-tools5
